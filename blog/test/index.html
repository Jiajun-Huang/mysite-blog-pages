Python
# Hello World in multiple languages
# 多言語でのHello World
def hello_world():
    languages = {
        'English': 'Hello, World!',
        '日本語': 'こんにちは、世界！',
        '한국어': '안녕하세요, 세계!',
        '中文': '你好，世界！'
    }

    for lang, greeting in languages.items():
        print(f"{lang}: {greeting}")

if __name__ == "__main__":
    hello_world()

JavaScript Code / JavaScript コード / 자바스크립트 코드 / JavaScript 代码

JavaScript
// Multi-language greeting function
// 多言語挨拶関数
function multiLanguageGreeting() {
  const greetings = {
    English: 'Hello, World!',
    日本語: 'こんにちは、世界！',
    한국어: '안녕하세요, 세계!',
    中文: '你好，世界！',
  };

  Object.entries(greetings).forEach(([lang, greeting]) => {
    console.log(`${lang}: ${greeting}`);
  });
}

multiLanguageGreeting();
